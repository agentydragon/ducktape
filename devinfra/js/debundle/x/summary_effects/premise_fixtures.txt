export function pure(x) { return { x }; }
export function snapshot(o) { return { ...o }; }
export function factory(fn) { return { fn }; }
export function invoker(fn) { return fn(); }
export function decorate(decorators, target, key) { let r = target; for (const d of decorators) r = d(target, key) || r; return r; }
let mode = 'a'; let count = 0;
export function setMode(m) { mode = m; }
export function tag(target) { target.tag = mode; }
export function next() { return ++count; }
export class Box { constructor(v) { this.v = v; } }
